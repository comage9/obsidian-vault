# 리눅스 헤르메스 봇 데스크톱 설치 지시서

## 개요
우분투(리눅스) 환경에서 실행되는 Hermes Agent 인스턴스 설치 및 설정 가이드.
KI AI 트레이더 전담 봇을 포함한 다중 봇 운영 환경 구축.

## 시스템 요구사항

- Python 3.11+
- PostgreSQL 14+ (로컬)
- Redis 7+ (선택, 메모리 캐싱용)
- 메모리 8GB 이상 권장 (트레이딩 봇 동시 실행 시)

## 3. KI AI 트레이더 전담 봇 설정

### 봇 실행 스크립트 생성
```bash
cat > /home/comtop/workspace/ki-ai-trader/run_ki_trader.py <<'EOF'
#!/usr/bin/env python3
"""
KI AI Trader 전담 봇
- 실계좌 자동 매매
- Phase 1~8 안전장치 적용
- 5초 주기 실시간 모니터링
"""
import os
import sys
import asyncio
import signal
from pathlib import Path

# 프로젝트 루트 추가
PROJECT_ROOT = Path(__file__).parent
sys.path.insert(0, str(PROJECT_ROOT / "src"))

from run_arena_trader import MultiAgentTrader

class KITraderBot:
    def __init__(self):
        self.trader = None
        self.running = False
        
    async def start(self):
        """봇 시작"""
        print("🤖 KI AI Trader Bot 시작 중...")
        self.trader = MultiAgentTrader()
        self.running = True
        
        # 시그널 핸들러 등록
        signal.signal(signal.SIGTERM, self._signal_handler)
        signal.signal(signal.SIGINT, self._signal_handler)
        
        # 메인 루프 실행
        await self.trader.run()
        
    def _signal_handler(self, signum, frame):
        print(f"\n🛑 시그널 {signum} 수신 - 정상 종료 중...")
        self.running = False
        if self.trader:
            self.trader.shutdown()

async def main():
    bot = KITraderBot()
    await bot.start()

if __name__ == "__main__":
    asyncio.run(main())
EOF

chmod +x /home/comtop/workspace/ki-ai-trader/run_ki_trader.py
```

## 4. 추가 봇 생성 (필요 시)

### 봇 관리자 스크립트
```bash
cat > /home/comtop/workspace/ki-ai-trader/bot_manager.py <<'EOF'
#!/usr/bin/env python3
"""
다중 봇 관리자
- KI Trader (실계좌 매매)
- VF2 Bot (물류 자동화)
- Monitor Bot (시스템 감시)
"""
import asyncio
import subprocess
import signal
import sys
from pathlib import Path
from typing import Dict, Optional

class BotManager:
    def __init__(self):
        self.processes: Dict[str, subprocess.Popen] = {}
        self.project_root = Path(__file__).parent
        
    def start_bot(self, name: str, script: str, env: dict = None) -> bool:
        """봇 프로세스 시작"""
        if name in self.processes:
            print(f"⚠️ {name} 이미 실행 중")
            return False
            
        cmd = [
            str(self.project_root / ".venv-3.11" / "bin" / "python"),
            script
        ]
        
        full_env = os.environ.copy()
        full_env.update({
            "PYTHONPATH": str(self.project_root / "src"),
            **(env or {})
        })
        
        try:
            proc = subprocess.Popen(
                cmd,
                cwd=self.project_root,
                env=full_env,
                stdout=subprocess.PIPE,
                stderr=subprocess.STDOUT,
                text=True,
                bufsize=1
            )
            self.processes[name] = proc
            print(f"✅ {name} 시작됨 (PID: {proc.pid})")
            return True
        except Exception as e:
            print(f"❌ {name} 시작 실패: {e}")
            return False
            
    def stop_bot(self, name: str) -> bool:
        """봇 프로세스 중지"""
        if name not in self.processes:
            return False
            
        proc = self.processes[name]
        proc.terminate()
        try:
            proc.wait(timeout=10)
        except subprocess.TimeoutExpired:
            proc.kill()
            proc.wait()
        del self.processes[name]
        print(f"🛑 {name} 중지됨")
        return True
        
    def stop_all(self):
        """모든 봇 중지"""
        for name in list(self.processes.keys()):
            self.stop_bot(name)
            
    def status(self) -> Dict:
        """상태 확인"""
        return {
            name: {
                "pid": proc.pid,
                "running": proc.poll() is None,
                "returncode": proc.poll()
            }
            for name, proc in self.processes.items()
        }

# 사용 예시
async def main():
    manager = BotManager()
    
    # KI Trader 시작
    manager.start_bot("ki-trader", "run_ki_trader.py")
    
    # 모니터링 봇 (선택)
    # manager.start_bot("monitor", "monitor_bot.py")
    
    # 상태 확인
    while True:
        await asyncio.sleep(30)
        status = manager.status()
        for name, info in status.items():
            if not info["running"]:
                print(f"⚠️ {name} 종료됨 (code: {info['returncode']}) - 재시작")
                manager.start_bot(name, f"run_{name}.py")

if __name__ == "__main__":
    import os
    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print("\n🛑 전체 중지...")
        BotManager().stop_all()
EOF
```

## 7. 문제 해결

### 자주 발생하는 이슈

| 이슈 | 원인 | 해결 |
|------|------|------|
| Exit 137 (OOM) | 메모리 부족 | MemoryMax=2G 설정, gc.collect() 주기적 호출 |
| Exit 1 | PYTHONPATH 불일치 | systemd Environment=PYTHONPATH 설정 |
| 포트 충돌 (8000) | 기존 프로세스 잔존 | `lsof -ti:8000 \| xargs kill -9` |
| 토큰 만료 | 1시간 만료 | AccountModeManager 자동 갱신 확인 |
| DB 연결 실패 | PostgreSQL 미실행 | `systemctl status postgresql` 확인 |
